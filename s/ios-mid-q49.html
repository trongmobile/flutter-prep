ViewModel có debounce 300ms cho ô search, logic "mã giảm giá hết hạn cuối ngày" và retry tối đa 3 lần với delay tăng dần. Test hiện dùng Thread.sleep hoặc expectation chờ vài giây nên chậm và đôi khi fail, còn test hết hạn pass ở máy dev nhưng fail ở CI đặt múi giờ khác. Bạn viết lại test thế nào?

Tự trả lời thành tiếng trước, rồi mới xem đáp án mẫu nhé.

Xem đáp án mẫu
Gốc rễ là code gọi thẳng các nguồn không điều khiển được: Date(), Task.sleep, Timer, UUID(), Calendar.current/TimeZone.current. Giải pháp là inject chúng ở ranh giới ViewModel hay service. Thời điểm: nhận now: () -> Date rồi test trả một mốc cố định. Chờ đợi: nhận any Clock<Duration> (production dùng ContinuousClock()) và gọi clock.sleep(for:); test dùng clock giả có advance(by:) (thư viện swift-clocks của Point-Free có sẵn TestClock) để debounce 300ms chạy xong trong mili-giây thay vì chờ thật. Retry với delay tăng dần thì nhận một closure sleep và assert danh sách delay đã được yêu cầu, không chờ thật. Lịch: đừng dùng .current; cố định Calendar với timeZone và Locale rõ ràng, nếu không test pass ở máy dev nhưng fail ở CI đặt múi giờ khác. Test các mốc biên như 23:59:59 và 00:00:00, cả ngày chuyển DST nếu app có thị trường dùng DST. UUID() gọi trực tiếp trong code cũng nên thành makeID: () -> UUID. Đừng inject mọi thứ ở mọi tầng, chỉ ở nơi logic thật sự phụ thuộc thời gian; còn Thread.sleep hay expectation chờ vài giây trong test chỉ che lấp vấn đề, làm CI chậm và flaky.
struct CouponValidator {
    var now: () -> Date = Date.init
    var calendar: Calendar

    // Mã dùng được hết ngày lastValidDay theo múi giờ của `calendar`
    func isExpired(_ coupon: Coupon) -> Bool {
        let startOfDay = calendar.startOfDay(for: coupon.lastValidDay)
        guard let nextDay = calendar.date(byAdding: .day, value: 1, to: startOfDay) else { return true }
        return now() >= nextDay
    }
}

@Test func couponExpiresAtEndOfDay() {
    var cal = Calendar(identifier: .gregorian)
    cal.timeZone = TimeZone(identifier: "Asia/Ho_Chi_Minh")! // cố định, không phụ thuộc máy chạy test
    let lastDay = cal.date(from: DateComponents(year: 2026, month: 3, day: 1))!
    let justBefore = cal.date(from: DateComponents(year: 2026, month: 3, day: 1, hour: 23, minute: 59, second: 59))!
    let midnight = cal.date(from: DateComponents(year: 2026, month: 3, day: 2))!
    let coupon = Coupon(lastValidDay: lastDay)

    #expect(!CouponValidator(now: { justBefore }, calendar: cal).isExpired(coupon))
    #expect(CouponValidator(now: { midnight }, calendar: cal).isExpired(coupon))
}

// Retry: nhận closure sleep thay vì gọi Task.sleep thật
func retry<T>(maxAttempts: Int = 3,
              sleep: (Duration) async throws -> Void = { try await Task.sleep(for: $0) },
              _ operation: () async throws -> T) async throws -> T {
    var attempt = 0
    while true {
        do { return try await operation() } catch {
            attempt += 1
            if attempt >= maxAttempts { throw error }
            try await sleep(.seconds(1 << (attempt - 1))) // 1s, 2s, ...
        }
    }
}

@Test func retryBacksOffWithoutWaiting() async throws {
    var delays: [Duration] = []
    var calls = 0
    let value = try await retry(sleep: { delays.append($0) }) {
        calls += 1
        if calls < 3 { throw URLError(.timedOut) }
        return "ok"
    }
    #expect(value == "ok")
    #expect(delays == [.seconds(1), .seconds(2)]) // kiểm tra delay đã yêu cầu, test chạy tức thì
}
