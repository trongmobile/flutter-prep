Design system cần một TagFlow xếp các chip theo hàng và xuống dòng khi hết chỗ, và một hàng nút có chiều rộng bằng nút rộng nhất. Layout của SwiftUI hoạt động theo quy tắc nào, và bạn viết container tuỳ biến bằng protocol Layout thế nào?

Tự trả lời thành tiếng trước, rồi mới xem đáp án mẫu nhé.

Xem đáp án mẫu
Layout của SwiftUI có ba bước: parent đề xuất một kích thước cho con (ProposedViewSize), con tự chọn kích thước của mình (có thể khác đề xuất), rồi parent đặt con vào vị trí. Đề xuất có thể là nil (kích thước lý tưởng của view), 0 (nhỏ nhất) hoặc .infinity (lớn nhất), nhờ đó parent hỏi được kích thước tự nhiên, nhỏ nhất và lớn nhất của một view. Protocol Layout (iOS 16 trở lên) cho phép tự viết container theo đúng quy trình đó: sizeThatFits(proposal:subviews:cache:) trả kích thước của container và placeSubviews(in:proposal:subviews:cache:) đặt từng subview bằng place(at:anchor:proposal:). subviews[i].sizeThatFits(...) hỏi con kích thước ứng với một đề xuất và có thể gọi nhiều lần với các đề xuất khác nhau, nhưng mỗi lần đều tốn chi phí nên dùng cache (makeCache, updateCache) để lưu kết quả đã đo giữa sizeThatFits và placeSubviews. Với hàng nút bằng nhau, hỏi từng con theo đề xuất .unspecified, lấy chiều rộng lớn nhất rồi đề xuất đúng chiều rộng đó cho mọi con. Con có thể truyền thông tin riêng cho layout (độ ưu tiên, nhãn) qua LayoutValueKey. AnyLayout cho phép đổi giữa hai layout (ngang hay dọc theo Dynamic Type) mà vẫn giữ identity nên state và animation của view không bị mất. ViewThatFits chọn view đầu tiên vừa chỗ, hợp khi chỉ cần một phương án dự phòng. Đừng dùng GeometryReader kèm PreferenceKey để tự ghép kích thước khi Layout giải quyết được, vì chúng thêm một lượt layout và dễ gây vòng lặp cập nhật.
struct TagFlow: Layout {
  var spacing: CGFloat = 8

  func sizeThatFits(proposal: ProposedViewSize, subviews: Subviews, cache: inout ()) -> CGSize {
    arrange(in: proposal.width ?? .infinity, subviews: subviews).size
  }

  func placeSubviews(in bounds: CGRect, proposal: ProposedViewSize, subviews: Subviews, cache: inout ()) {
    let result = arrange(in: bounds.width, subviews: subviews)
    for (subview, origin) in zip(subviews, result.origins) {
      subview.place(at: CGPoint(x: bounds.minX + origin.x, y: bounds.minY + origin.y), proposal: .unspecified)
    }
  }

  // Tính vị trí các chip; có thể lưu vào cache để tránh tính hai lần
  private func arrange(in width: CGFloat, subviews: Subviews) -> (size: CGSize, origins: [CGPoint]) {
    var origins: [CGPoint] = []
    var x: CGFloat = 0
    var y: CGFloat = 0
    var rowHeight: CGFloat = 0
    var maxX: CGFloat = 0
    for subview in subviews {
      let size = subview.sizeThatFits(.unspecified) // kích thước tự nhiên của chip
      if x > 0 && x + size.width > width { // hết chỗ: xuống hàng
        x = 0
        y += rowHeight + spacing
        rowHeight = 0
      }
      origins.append(CGPoint(x: x, y: y))
      x += size.width + spacing
      maxX = max(maxX, x - spacing)
      rowHeight = max(rowHeight, size.height)
    }
    return (CGSize(width: maxX, height: y + rowHeight), origins)
  }
}

// Dùng như một container bình thường
TagFlow(spacing: 8) {
  ForEach(tags) { TagChip($0) }
}

// Đổi layout theo Dynamic Type mà vẫn giữ identity và animation
@Environment(\.dynamicTypeSize) private var typeSize
let layout = typeSize.isAccessibilitySize ? AnyLayout(VStackLayout()) : AnyLayout(HStackLayout())
layout {
  Text("Giá")
  Text("1.000 ₫")
}
