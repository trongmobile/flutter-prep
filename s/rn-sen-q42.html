Team chuyển sang trunk-based và muốn mọi tính năng lớn nằm sau feature flag, kèm kill-switch tắt được trong vài phút mà không cần phát hành bản mới. Bạn thiết kế hệ thống flag/remote config phía app RN thế nào?

Tự trả lời thành tiếng trước, rồi mới xem đáp án mẫu nhé.

Xem đáp án mẫu
Trước hết phân loại flag vì vòng đời khác nhau: release flag (tạm thời, xoá sau khi bật 100%), kill-switch (vận hành, tồn tại lâu), experiment (A/B). Nguồn giá trị có thể là dịch vụ sẵn có (LaunchDarkly, Firebase Remote Config, GrowthBook...) hoặc endpoint tự dựng; điều quan trọng là cách app tiêu thụ. Một: luôn có giá trị mặc định biên dịch sẵn trong app (tính năng rủi ro mặc định tắt) để app chạy đúng khi mất mạng hoặc server lỗi. Hai: lúc mở app đọc đồng bộ snapshot đã cache (MMKV đọc được đồng bộ) nên render đầu tiên đã ổn định; tải bản mới ở nền với timeout ngắn, không chặn splash. Ba: tránh flicker và trạng thái nửa vời: giá trị mới chỉ áp dụng từ lần mở sau hoặc tại điểm an toàn; riêng kill-switch có hiệu lực ngay và UI phải thoát êm. Bốn: chia rollout theo phần trăm bằng bucket ổn định từ userId (không random mỗi phiên) và theo phiên bản app tối thiểu, vì bản cũ không có code của tính năng; với OTA, code mới đi cùng flag tắt rồi bật sau, nhưng JS vẫn phải tương thích với native hiện có. Năm: kill-switch không được phụ thuộc vào thứ đang hỏng; nếu app crash lúc khởi động nhiều lần liên tiếp thì vào chế độ an toàn dùng giá trị mặc định. Sáu: kiểm soát nợ flag bằng registry có kiểu TypeScript, người sở hữu và ngày hết hạn, CI báo flag quá hạn; gắn các flag đang bật vào crash report để đối chiếu lỗi. Chỉ test hai nhánh của flag quan trọng, đừng nhân tổ hợp.
const DEFAULTS = { newCheckout: false, killVideoPreload: false };
type FlagName = keyof typeof DEFAULTS;
const KILL_SWITCHES: FlagName[] = ['killVideoPreload'];

// storage là MMKV (đọc đồng bộ): snapshot đầu tiên có ngay, UI không bị nháy
let active: Record<FlagName, boolean> = { ...DEFAULTS, ...JSON.parse(storage.getString('flags') ?? '{}') };
const listeners = new Set<() => void>();

export async function refreshFlags(): Promise<void> {
  try {
    const remote: Partial<Record<FlagName, boolean>> = await fetchJson(FLAGS_URL, { timeoutMs: 3000 });
    storage.set('flags', JSON.stringify(remote)); // flag thường: áp dụng từ lần mở app sau

    let changed = false;
    for (const key of KILL_SWITCHES) {
      const value = remote[key];
      if (value !== undefined && value !== active[key]) {
        active = { ...active, [key]: value }; // kill-switch có hiệu lực ngay trong phiên
        changed = true;
      }
    }
    if (changed) listeners.forEach((notify) => notify());
  } catch {
    // mất mạng hoặc server lỗi: giữ nguyên giá trị cache/mặc định
  }
}

export function useFlag(name: FlagName): boolean {
  return useSyncExternalStore(
    (onChange) => {
      listeners.add(onChange);
      return () => listeners.delete(onChange);
    },
    () => active[name],
  );
}
