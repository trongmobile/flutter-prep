App bán hàng offline-first cần lưu khoảng 300.000 bản ghi (sản phẩm, đơn nháp) và tìm kiếm trên đó. Bản hiện tại lưu một JSON lớn trong AsyncStorage nên mở app chậm vài giây và thỉnh thoảng mất dữ liệu. Bạn chọn giải pháp lưu trữ nào, và xử lý mã hoá, migration, hiệu năng ra sao?

Tự trả lời thành tiếng trước, rồi mới xem đáp án mẫu nhé.

Xem đáp án mẫu
Vấn đề gốc là dùng key-value cho dữ liệu quan hệ: mỗi lần đọc hay ghi phải parse hoặc serialize cả khối JSON trên JS thread, app bị kill giữa lúc ghi dễ hỏng dữ liệu, và không có index. Phân loại theo nhu cầu: AsyncStorage (bất đồng bộ, có giới hạn dung lượng mặc định trên Android) cho vài giá trị nhỏ; MMKV (đồng bộ, nhanh nhờ JSI, có tuỳ chọn mã hoá) rất hợp cho cài đặt, flag, session nhưng vẫn là key-value, không truy vấn được; SQLite (op-sqlite, expo-sqlite) cho dữ liệu hàng trăm nghìn dòng: index, truy vấn có điều kiện và phân trang, giao dịch nguyên tử, tìm kiếm văn bản nếu bật FTS, và SQLCipher khi cần mã hoá cả file (một số thư viện hỗ trợ bản build SQLCipher). WatermelonDB xây trên SQLite với truy vấn quan sát được và nạp lười, hợp danh sách phản ứng theo dữ liệu nhưng thêm lớp model và cấu hình; ORM kiểu Drizzle cho truy vấn có kiểu trên SQLite. Với các CSDL khác (Realm...) hãy kiểm tra tình trạng bảo trì trước khi gắn chặt. Migration: đánh số phiên bản schema (PRAGMA user_version), mỗi bước chạy trong một transaction cùng việc tăng phiên bản, không bao giờ xoá dữ liệu im lặng, và test migrate từ fixture của mọi bản đã phát hành. Hiệu năng: tạo index cho cột tìm kiếm, nhập hàng loạt trong một transaction (nhanh hơn nhiều so với ghi từng dòng), chỉ truy vấn phần cần hiển thị với LIMIT thay vì kéo cả bảng vào mảng JS rồi filter, và để ảnh hay file lớn ở hệ thống file chứ không nhét vào DB.
import * as SQLite from 'expo-sqlite';

const MIGRATIONS = [
  `CREATE TABLE product (id INTEGER PRIMARY KEY, sku TEXT NOT NULL, name TEXT NOT NULL, price INTEGER NOT NULL);
   CREATE INDEX idx_product_name ON product(name COLLATE NOCASE);`,
  'ALTER TABLE product ADD COLUMN barcode TEXT;',
];

export async function openDb() {
  const db = await SQLite.openDatabaseAsync('shop.db');
  await db.execAsync('PRAGMA journal_mode = WAL;');
  const row = await db.getFirstAsync<{ user_version: number }>('PRAGMA user_version');
  for (let v = row?.user_version ?? 0; v < MIGRATIONS.length; v++) {
    await db.withTransactionAsync(async () => {
      await db.execAsync(MIGRATIONS[v]);
      await db.execAsync(`PRAGMA user_version = ${v + 1}`); // tăng phiên bản cùng transaction với bước migrate
    });
  }
  return db;
}

// Nhập hàng loạt trong một transaction
export async function importProducts(db: SQLite.SQLiteDatabase, items: Product[]) {
  await db.withTransactionAsync(async () => {
    for (const p of items) {
      await db.runAsync('INSERT OR REPLACE INTO product (id, sku, name, price) VALUES (?, ?, ?, ?)', [p.id, p.sku, p.name, p.price]);
    }
  });
}

// Chỉ lấy 50 dòng đầu; LIKE khớp đầu chuỗi mới tận dụng được index (tìm giữa chuỗi nên dùng FTS)
export const searchProducts = (db: SQLite.SQLiteDatabase, term: string) =>
  db.getAllAsync<Product>('SELECT id, sku, name, price FROM product WHERE name LIKE ? ORDER BY name LIMIT 50', [`${term}%`]);
