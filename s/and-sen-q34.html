Bạn cần đổi schema Room (thêm cột NOT NULL, tách dữ liệu một cột sang bảng mới) cho app có vài triệu user đang ở nhiều phiên bản cũ khác nhau. Bạn viết và kiểm thử migration thế nào, và vì sao fallbackToDestructiveMigration nguy hiểm?

Tự trả lời thành tiếng trước, rồi mới xem đáp án mẫu nhé.

Xem đáp án mẫu
Mỗi lần đổi schema phải tăng version và có đường nâng cấp từ mọi version cũ còn người dùng, vì user có thể nhảy từ bản rất cũ lên thẳng bản mới nhất. Thay đổi đơn giản (thêm cột, thêm bảng, đổi tên) dùng AutoMigration kèm exportSchema = true và commit thư mục schema JSON vào git để Room có bản so sánh; thay đổi phức tạp như tách bảng hay chuyển dữ liệu thì viết Migration bằng SQL. Cột NOT NULL mới phải có DEFAULT để các dòng cũ hợp lệ, và entity phải khai báo @ColumnInfo(defaultValue = ...) cho khớp, nếu không Room báo schema sai khi validate. SQLite trên máy cũ không hỗ trợ xoá hay đổi kiểu cột tuỳ ý, nên đổi lớn thường theo cách tạo bảng mới, copy dữ liệu, xoá bảng cũ, đổi tên. Tôi theo kiểu expand–contract: bản này thêm bảng mới và copy dữ liệu, giữ nguyên cột cũ; bản sau khi chắc chắn mới xoá cột cũ. Kiểm thử bằng MigrationTestHelper: tạo DB ở version cũ, chèn dữ liệu, chạy runMigrationsAndValidate rồi kiểm tra dữ liệu, và test cả chuỗi nhiều bước. fallbackToDestructiveMigration() xoá sạch dữ liệu local khi thiếu đường nâng cấp; với cache thì chấp nhận được, nhưng với dữ liệu chỉ có trên máy (nháp, outbox chưa đồng bộ) thì là mất dữ liệu của user và bạn chỉ biết sau khi đã release. Migration chạy ở lần truy cập DB đầu tiên nên phải nhanh, đo với dữ liệu thật lớn và đảm bảo lần mở đó không nằm trên main thread.
@Database(
  entities = [OrderEntity::class, OrderNoteEntity::class],
  version = 3,
  exportSchema = true, // commit thư mục schema vào git để Room so sánh và test migration
)
abstract class AppDatabase : RoomDatabase()

val MIGRATION_2_3 = object : Migration(2, 3) {
  override fun migrate(db: SupportSQLiteDatabase) {
    // Cột NOT NULL mới phải có DEFAULT thì các dòng cũ mới hợp lệ
    db.execSQL("ALTER TABLE orders ADD COLUMN status TEXT NOT NULL DEFAULT 'PENDING'")
    // Expand: tạo bảng mới và copy dữ liệu, giữ nguyên cột note, bản sau mới xoá
    db.execSQL("CREATE TABLE IF NOT EXISTS order_notes (orderId TEXT NOT NULL, body TEXT NOT NULL, PRIMARY KEY(orderId))")
    db.execSQL("INSERT INTO order_notes (orderId, body) SELECT id, note FROM orders WHERE note IS NOT NULL")
  }
}

@Test
fun migrate2To3_keepsNotes() {
  helper.createDatabase(TEST_DB, 2).apply {
    execSQL("INSERT INTO orders (id, note) VALUES ('a1', 'giao giờ hành chính')")
    close()
  }
  val db = helper.runMigrationsAndValidate(TEST_DB, 3, true, MIGRATION_2_3) // so với schema đã export
  db.query("SELECT body FROM order_notes WHERE orderId = 'a1'").use {
    assertTrue(it.moveToFirst() && it.getString(0) == "giao giờ hành chính")
  }
}
