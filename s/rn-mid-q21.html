Dự án Expo cần tích hợp một SDK yêu cầu thêm meta-data vào AndroidManifest.xml. Với Continuous Native Generation (expo prebuild), bạn làm thế nào thay vì sửa tay thư mục native?

Tự trả lời thành tiếng trước, rồi mới xem đáp án mẫu nhé.

Xem đáp án mẫu
Với CNG, thư mục android/ios được coi là sản phẩm sinh ra từ app.json/app.config.ts cùng các config plugin và thường không commit; npx expo prebuild --clean có thể xoá đi sinh lại, nên mọi sửa tay đều mất. Nếu thư viện có sẵn config plugin thì chỉ cần thêm vào mảng plugins kèm option. Nếu không, tự viết config plugin bằng các mod như withAndroidManifest, withInfoPlist, withAppDelegate để chỉnh file native một cách lặp lại được và review được. Lưu ý app có native code tuỳ biến thì không chạy trong Expo Go nữa, cần development build (qua EAS Build hoặc npx expo run:android).
// plugins/withMapsApiKey.ts
import { ConfigPlugin, withAndroidManifest, AndroidConfig } from 'expo/config-plugins';

const withMapsApiKey: ConfigPlugin<{ apiKey: string }> = (config, { apiKey }) =>
  withAndroidManifest(config, (cfg) => {
    const app = AndroidConfig.Manifest.getMainApplicationOrThrow(cfg.modResults);
    AndroidConfig.Manifest.addMetaDataItemToMainApplication(
      app, 'com.google.android.geo.API_KEY', apiKey,
    );
    return cfg;
  });

export default withMapsApiKey;
