Màn thanh toán: người dùng bấm "Đặt hàng" hai lần liên tiếp, hoặc bấm một lần nhưng mạng đứt đúng lúc request đã tới server rồi họ bấm lại. Kết quả là hai đơn trùng và bị trừ tiền hai lần. Bạn thiết kế phía app thế nào để không tạo đơn trùng?

Tự trả lời thành tiếng trước, rồi mới xem đáp án mẫu nhé.

Xem đáp án mẫu
Cần hai lớp. Lớp UI chỉ giảm bớt: đổi sang trạng thái submitting và disable nút ngay khi bấm; nhưng setState chưa kịp render thì lần bấm thứ hai vẫn lọt vào, nên thêm một useRef làm khoá đồng bộ (if (inFlight.current) return). Lớp quyết định là idempotency key: app sinh một UUID cho mỗi "ý định đặt hàng" (một giỏ hàng ở màn thanh toán) và gửi trong header Idempotency-Key; server lưu key cùng kết quả, nên cùng key gửi lại thì trả đơn cũ chứ không tạo mới. Key phải giữ nguyên qua mọi lần retry và chỉ đổi khi người dùng sửa giỏ hàng; sinh key mới mỗi lần bấm là sai vì server không nhận ra đó là cùng một đơn. Khi timeout hoặc mất mạng, kết quả là "chưa rõ" chứ không phải "thất bại": hiện trạng thái "Đang xác nhận đơn hàng…", cho thử lại với cùng key hoặc hỏi server trạng thái đơn theo key, và chưa cho tạo đơn mới. Nếu app bị kill giữa chừng thì lưu key và trạng thái pending vào storage để lần mở sau kiểm tra lại. Chỉ retry lỗi mạng và 5xx, kèm backoff, không retry 4xx. Tính idempotent do server bảo đảm; app có nhiệm vụ gửi đúng key.
function CheckoutButton({ cart }: { cart: Cart }) {
  const [status, setStatus] = useState<'idle' | 'submitting' | 'unknown' | 'done'>('idle');
  const inFlight = useRef(false); // khoá đồng bộ: setState chưa kịp render thì lần bấm 2 vẫn lọt
  // Một "ý định đặt hàng": giữ nguyên qua các lần retry. Đổi giỏ hàng thì remount (key={cart.version}) để có key mới
  const [idempotencyKey] = useState(() => uuid());

  async function submit() {
    if (inFlight.current) return;
    inFlight.current = true;
    setStatus('submitting');
    try {
      await api.post('/orders', cart, { headers: { 'Idempotency-Key': idempotencyKey } });
      setStatus('done');
    } catch (e) {
      // Timeout/mất mạng: chưa biết server đã tạo đơn hay chưa -> 'unknown', thử lại dùng đúng key cũ
      setStatus(isNetworkError(e) ? 'unknown' : 'idle');
    } finally {
      inFlight.current = false;
    }
  }

  return (
    <Button
      title={status === 'unknown' ? 'Thử lại' : 'Đặt hàng'}
      onPress={submit}
      disabled={status === 'submitting' || status === 'done'}
    />
  );
}
